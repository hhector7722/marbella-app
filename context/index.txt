/**
 * Ventana de negocio BDP (Data Sistema). Funciona aunque Fecha (día TPV) vaya desfasada.
 * Ventas: tickets con Hora_Cierre (finalizados) o Pendiente=1 (a cuenta, sin cierre).
 * Tickets abiertos sin pendiente quedan fuera (radar sala sigue por Comandas / DIRECTO reciente).
 */
const VENTAS_WHERE = `
  WHERE (
    (Fecha_Sistema IS NOT NULL AND CAST(Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE)))
    OR (Fecha_Sistema IS NULL AND CAST(Fecha AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE)))
  )
  AND (Hora_Cierre IS NOT NULL OR Pendiente = 1)
`;

const CABECERA_SELECT = `
  Numero_Documento, Serie, Numero, Total_Documento,
  Fecha, Fecha_Sistema, Hora_Cierre,
  ISNULL(Mesa, 0) AS Mesa,
  ISNULL(Pendiente, 0) AS Pendiente
`;

function webhookHeaders() {
    const headers = {};
    if (process.env.WEBHOOK_SECRET) {
        headers.Authorization = `Bearer ${process.env.WEBHOOK_SECRET}`;
    }
    return headers;
}

function ticketKey(numeroDocumento) {
    return String(numeroDocumento ?? '').trim();
}

function roundMoney(n) {
    return Math.round((Number(n) || 0) * 100) / 100;
}

/** Verdad única UTC: siempre ISO-8601 terminado en Z (nunca Date crudo en JSON). */
function toIso(val) {
    if (val == null) return null;
    const d = new Date(val);
    if (Number.isNaN(d.getTime())) return null;
    const iso = d.toISOString();
    return iso.endsWith('Z') ? iso : `${iso}Z`;
}

function toYmdLocal(val) {
    if (val == null) return null;
    const d = new Date(val);
    if (Number.isNaN(d.getTime())) return null;
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${y}-${m}-${day}`;
}

function ticketPaymentSignature(payload) {
    return [
        payload.cobro_efectivo,
        payload.cobro_tarjeta,
        payload.cobro_pendiente,
        payload.total_documento,
    ].join('|');
}

async function buildPayloadVenta(pool, t) {
    const pagosRes = await pool
        .request()
        .input('s', t.Serie)
        .input('n', t.Numero)
        .query(`
            SELECT
                ISNULL(SUM(CASE WHEN Forma_Pago = 1 THEN Importe_Pago ELSE 0 END), 0) AS efectivo,
                ISNULL(SUM(CASE WHEN Forma_Pago IN (2, 3) THEN Importe_Pago ELSE 0 END), 0) AS tarjeta
            FROM Documentos_Pagos WITH (NOLOCK)
            WHERE Serie = @s AND Numero = @n
        `);

    const lRes = await pool
        .request()
        .input('s', t.Serie)
        .input('n', t.Numero)
        .query(
            'SELECT PLU, Descripcion, Unidades, Precio, Hora FROM Documentos_Lineas WITH (NOLOCK) WHERE Serie = @s AND Numero = @n'
        );
    const mRes = await pool
        .request()
        .input('s', t.Serie)
        .input('n', t.Numero)
        .query(
            'SELECT PLU, Descripcion, Unidades FROM Documentos_Menus WITH (NOLOCK) WHERE Serie = @s AND Numero = @n'
        );
    const pRes = await pool
        .request()
        .input('s', t.Serie)
        .input('n', t.Numero)
        .query(
            'SELECT PLU, Descripcion, Unidades FROM Documentos_Packs WITH (NOLOCK) WHERE Serie = @s AND Numero = @n'
        );

    const horaLinea = lRes.recordset[0]?.Hora;
    const total = parseFloat(t.Total_Documento) || 0;
    const cobro_efectivo = roundMoney(parseFloat(pagosRes.recordset[0]?.efectivo) || 0);
    const cobro_tarjeta = roundMoney(parseFloat(pagosRes.recordset[0]?.tarjeta) || 0);
    const pendienteBit = t.Pendiente === true || t.Pendiente === 1;

    let cobro_pendiente = roundMoney(Math.max(0, total - cobro_efectivo - cobro_tarjeta));
    if (pendienteBit && cobro_efectivo === 0 && cobro_tarjeta === 0) {
        cobro_pendiente = roundMoney(total);
    }

    const horaCierreIso =
        toIso(t.Hora_Cierre) || (pendienteBit ? toIso(t.Fecha_Sistema) || toIso(t.Fecha) : null);
    const fechaTicketIso = horaLinea ? toIso(horaLinea) : horaCierreIso || toIso(new Date());

    return {
        numero_documento: ticketKey(t.Numero_Documento),
        mesa: t.Mesa,
        total_documento: total,
        cobro_efectivo,
        cobro_tarjeta,
        cobro_pendiente,
        fecha_sistema: toIso(t.Fecha_Sistema),
        hora_cierre: horaCierreIso,
        fecha_bdp: toIso(t.Fecha),
        fecha: fechaTicketIso,
        lineas: [
            ...lRes.recordset.map((l) => ({
                articulo_id: l.PLU,
                nombre: l.Descripcion?.trim() || 'Sin Nombre',
                unidades: parseFloat(l.Unidades) || 0,
                precio: parseFloat(l.Precio) || 0,
            })),
            ...mRes.recordset.map((m) => ({
                articulo_id: m.PLU,
                nombre: '↳ ' + (m.Descripcion?.trim() || 'Sin Nombre'),
                unidades: parseFloat(m.Unidades) || 0,
                precio: 0,
            })),
            ...pRes.recordset.map((p) => ({
                articulo_id: p.PLU,
                nombre: '↳ ' + (p.Descripcion?.trim() || 'Sin Nombre'),
                unidades: parseFloat(p.Unidades) || 0,
                precio: 0,
            })),
        ],
    };
}

async function enviarTicket(pool, t) {
    const key = ticketKey(t.Numero_Documento);
    const payload = await buildPayloadVenta(pool, t);
    const sig = ticketPaymentSignature(payload);

    if (memoriaTickets.has(key) && memoriaTickets.get(key) === sig) {
        return false;
    }

    await axios.post(
        URL_VENTAS,
        { ventas: [payload] },
        { timeout: AXIOS_TIMEOUT_MS, headers: webhookHeaders() }
    );
    memoriaTickets.set(key, sig);
    const tagCuenta =
        !payload.hora_cierre && payload.cobro_pendiente > 0 ? ' | A-CUENTA' : '';
    console.log(
        `✅ Venta enviada -> Ticket: ${key} | ef=${payload.cobro_efectivo} | tj=${payload.cobro_tarjeta} | pend=${payload.cobro_pendiente}${tagCuenta}`
    );
    return true;
}

async function catchUpVentas(pool) {
    console.log('📦 Catch-up ventas (Fecha_Sistema, paginado)...');
    let offset = 0;
    let enviados = 0;
    let escaneados = 0;

    while (true) {
        const res = await pool
            .request()
            .input('off', sql.Int, offset)
            .input('lim', sql.Int, CATCHUP_PAGE_SIZE)
            .query(`
                SELECT ${CABECERA_SELECT}
                FROM Documentos_Cabecera WITH (NOLOCK)
                ${VENTAS_WHERE}
                ORDER BY COALESCE(Hora_Cierre, Fecha_Sistema, Fecha) DESC
                OFFSET @off ROWS FETCH NEXT @lim ROWS ONLY
            `);

        const batch = res.recordset || [];
        if (!batch.length) break;

        for (const t of batch) {
            escaneados++;
            try {
                const sent = await enviarTicket(pool, t);
                if (sent) enviados++;
            } catch (e) {
                console.error(`❌ Catch-up ticket ${t.Numero_Documento}:`, e.message);
            }
            await new Promise((r) => setTimeout(r, 50));
        }

        offset += CATCHUP_PAGE_SIZE;
        if (batch.length < CATCHUP_PAGE_SIZE) break;
    }

    console.log(`📦 Catch-up listo: ${enviados} enviados (${escaneados} cabeceras en ventana).`);
}

async function logDiagnosticoSql(pool) {
    try {
        const diag = await pool.request().query(`
            SELECT
                GETDATE() AS sql_ahora,
                (SELECT COUNT(*) FROM Documentos_Cabecera WITH (NOLOCK) ${VENTAS_WHERE}) AS en_ventana,
                (SELECT COUNT(*) FROM Documentos_Cabecera WITH (NOLOCK) ${VENTAS_WHERE}
                    AND Pendiente = 1 AND Hora_Cierre IS NULL) AS pendiente_sin_cierre,
                (SELECT COUNT(*) FROM Documentos_Cabecera WITH (NOLOCK)
                    WHERE Hora_Cierre IS NOT NULL AND Hora_Cierre >= DATEADD(hour, -3, GETDATE())) AS hora_cierre_3h,
                (SELECT MAX(Hora_Cierre) FROM Documentos_Cabecera WITH (NOLOCK)) AS max_hora_cierre,
                (SELECT MAX(Numero_Documento) FROM Documentos_Cabecera WITH (NOLOCK) ${VENTAS_WHERE}) AS max_doc_ventana
        `);
        const d = diag.recordset[0] || {};
        console.log('🔎 SQL GETDATE()      =', d.sql_ahora);
        console.log('🔎 Docs en ventana    =', d.en_ventana, '(cerrados o a cuenta)');
        console.log('🔎 A cuenta sin cierre =', d.pendiente_sin_cierre);
        console.log('🔎 Docs Hora_Cierre 3h =', d.hora_cierre_3h);
        console.log('🔎 MAX Hora_Cierre    =', d.max_hora_cierre);
        console.log('🔎 MAX Num.Documento  =', d.max_doc_ventana);
    } catch (e) {
        console.error('🔎 Diagnóstico SQL falló:', e.message);
    }
}

async function pollVentasRecientes(pool) {
    if (ventasPollBusy) return;
    ventasPollBusy = true;
    try {
        const res = await pool.request().query(`
            SELECT TOP (${POLL_TOP}) ${CABECERA_SELECT}
            FROM Documentos_Cabecera WITH (NOLOCK)
            ${VENTAS_WHERE}
            ORDER BY
              CASE WHEN ISNULL(Pendiente, 0) = 1 AND Hora_Cierre IS NULL THEN 0 ELSE 1 END,
              COALESCE(Hora_Cierre, Fecha_Sistema, Fecha) DESC,
              Numero_Documento DESC
        `);

        const candidatos = res.recordset || [];
        let nuevos = 0;
        let actualizados = 0;

        for (const t of candidatos) {
            try {
                const sent = await enviarTicket(pool, t);
                if (sent) nuevos++;
                else actualizados++;
            } catch (e) {
                console.error(`❌ ERROR ticket ${ticketKey(t.Numero_Documento)}:`, e.message);
            }
        }

        pollTick++;
        const ts = new Date().toLocaleTimeString('es-ES', {
            hour: '2-digit',
            minute: '2-digit',
            second: '2-digit',
        });
        if (nuevos > 0) {
            console.log(
                `🔄 [${ts}] Poll ventas: ${nuevos} nuevo(s)/actualizado(s) | sin cambio=${actualizados} | candidatos=${candidatos.length} | memoria=${memoriaTickets.size}`
            );
        } else if (pollTick % 12 === 0) {
            console.log(
                `⏱ [${ts}] Poll ventas: 0 cambios | candidatos=${candidatos.length} | memoria=${memoriaTickets.size}`
            );
        }
    } finally {
        ventasPollBusy = false;
    }
}

async function pollCajaMovimientos(pool) {
    if (cajaPollBusy) return;
    cajaPollBusy = true;
    try {
        const res = await pool.request().query(`
            SELECT Fecha, Num_Concepto, Importe_Entrada
            FROM Caja_Lineas WITH (NOLOCK)
            WHERE Num_Concepto IN (107)
              AND Fecha >= DATEADD(hour, -24, GETDATE())
              AND ISNULL(Importe_Entrada, 0) <> 0
        `);

        const movimientos = (res.recordset || [])
            .map((row) => {
                const amount = roundMoney(row.Importe_Entrada);
                const movementDate = toIso(row.Fecha);
                const fechaNegocio = toYmdLocal(row.Fecha);
                const conceptCode = Number(row.Num_Concepto);
                if (!movementDate || !fechaNegocio || !Number.isFinite(conceptCode) || amount <= 0) {
                    return null;
                }
                return {
                    fecha_negocio: fechaNegocio,
                    movement_date: movementDate,
                    concept_code: conceptCode,
                    amount,
                    raw_json: {
                        Fecha: row.Fecha,
                        Num_Concepto: row.Num_Concepto,
                        Importe_Entrada: row.Importe_Entrada,
                    },
                };
            })
            .filter(Boolean);

        if (!movimientos.length) return;

        await axios.post(
            URL_CAJA,
            { movimientos },
            { timeout: AXIOS_TIMEOUT_MS, headers: webhookHeaders() }
        );
        console.log(`💶 Caja: ${movimientos.length} movimiento(s) concepto 107 enviados.`);
    } catch (e) {
        console.error('❌ Poll caja:', e.message);
    } finally {
        cajaPollBusy = false;
    }
}

async function start() {
    try {
        const pool = await sql.connect(config);
        console.log(`✅ Sistema Activado → ${DOMINIO}`);
        console.log(`📋 Poll ventas cada ${POLL_INTERVAL_MS / 1000}s | Poll caja cada ${CAJA_POLL_INTERVAL_MS / 1000}s`);
        console.log(`📋 Catch-up arranque: ${RUN_CATCHUP ? 'SÍ' : 'NO'}`);
        console.log(`📋 Caja → ${URL_CAJA}`);

        await logDiagnosticoSql(pool);

        // Recuperación: catch-up completo ANTES del poll (TOP 50 no basta para sincronizar el día).
        if (RUN_CATCHUP) {
            await catchUpVentas(pool);
        }

        void pollVentasRecientes(pool);
        void pollCajaMovimientos(pool);

        setInterval(async () => {
            try {
                await pollVentasRecientes(pool);
            } catch (e) {
                console.error('❌ ERROR EN VENTAS:', e.message);
            }
        }, POLL_INTERVAL_MS);

        setInterval(async () => {
            try {
                await pollCajaMovimientos(pool);
            } catch (e) {
                console.error('❌ ERROR EN CAJA:', e.message);
            }
        }, CAJA_POLL_INTERVAL_MS);

        // TELEMETRÍA / MONITOR COCINA
        setInterval(async () => {
            try {
                const resVivas = await pool.request().query(`
                    SELECT
                        CAST(C.Numero_Referencia AS VARCHAR) AS Numero_Referencia,
                        ISNULL(C.Mesa, 0) AS Mesa,
                        C.Nota_Cabecera1,
                        LTRIM(RTRIM(ISNULL(C.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        L.PLU as articulo_id,
                        L.Descripcion as nombre,
                        L.Unidades as unidades,
                        L.Precio as precio,
                        L.Hora,
                        LTRIM(RTRIM(ISNULL(L.Descripcion_Auxiliar, '') + ISNULL((SELECT ' [' + CC.Descripcion + ']' FROM Comandas_Comentarios CC WITH (NOLOCK) WHERE CC.Numero_Referencia = L.Numero_Referencia AND CC.Terminal = L.Terminal AND CC.Linea_Comandas = L.Linea FOR XML PATH('')), ''))) as notas
                    FROM Comandas_Cabecera C WITH (NOLOCK)
                    INNER JOIN Comandas_Lineas L WITH (NOLOCK) ON C.Numero_Referencia = L.Numero_Referencia AND C.Terminal = L.Terminal
                    WHERE L.Unidades <> 0

                    UNION ALL

                    SELECT
                        'DIRECTO-' + CAST(D.Numero AS VARCHAR) AS Numero_Referencia,
                        ISNULL(D.Mesa, 0) AS Mesa,
                        D.Nota_Cabecera1,
                        LTRIM(RTRIM(ISNULL(D.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        L.PLU as articulo_id,
                        L.Descripcion as nombre,
                        L.Unidades as unidades,
                        L.Precio as precio,
                        L.Hora,
                        ISNULL(L.Descripcion_Auxiliar, '') as notas
                    FROM Documentos_Cabecera D WITH (NOLOCK)
                    INNER JOIN Documentos_Lineas L WITH (NOLOCK) ON D.Serie = L.Serie AND D.Numero = L.Numero
                    WHERE L.Unidades <> 0
                    AND CAST(D.Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE))
                    AND D.Hora_Cierre >= DATEADD(minute, -15, GETDATE())
                    AND DATEDIFF(minute, D.Hora_Apertura, D.Hora_Cierre) <= 10

                    UNION ALL

                    SELECT
                        CAST(C.Numero_Referencia AS VARCHAR) AS Numero_Referencia,
                        ISNULL(C.Mesa, 0) AS Mesa,
                        C.Nota_Cabecera1,
                        LTRIM(RTRIM(ISNULL(C.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        M.PLU as articulo_id,
                        '↳ ' + LTRIM(RTRIM(M.Descripcion)) as nombre,
                        M.Unidades as unidades,
                        0 as precio,
                        NULL as Hora,
                        '' as notas
                    FROM Comandas_Cabecera C WITH (NOLOCK)
                    INNER JOIN Comandas_Menus M WITH (NOLOCK) ON C.Numero_Referencia = M.Numero_Referencia AND C.Terminal = M.Terminal
                    WHERE M.Unidades <> 0

                    UNION ALL

                    SELECT
                        'DIRECTO-' + CAST(D.Numero AS VARCHAR) AS Numero_Referencia,
                        ISNULL(D.Mesa, 0) AS Mesa,
                        D.Nota_Cabecera1,
                        LTRIM(RTRIM(ISNULL(D.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        M.PLU as articulo_id,
                        '↳ ' + LTRIM(RTRIM(M.Descripcion)) as nombre,
                        M.Unidades as unidades,
                        0 as precio,
                        NULL as Hora,
                        '' as notas
                    FROM Documentos_Cabecera D WITH (NOLOCK)
                    INNER JOIN Documentos_Menus M WITH (NOLOCK) ON D.Serie = M.Serie AND D.Numero = M.Numero
                    WHERE M.Unidades <> 0
                    AND CAST(D.Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE))
                    AND D.Hora_Cierre >= DATEADD(minute, -15, GETDATE())
                    AND DATEDIFF(minute, D.Hora_Apertura, D.Hora_Cierre) <= 10

                    UNION ALL

                    SELECT
                        CAST(C.Numero_Referencia AS VARCHAR) AS Numero_Referencia,
                        ISNULL(C.Mesa, 0) AS Mesa,
                        C.Nota_Cabecera1,
                        LTRIM(RTRIM(ISNULL(C.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        P.PLU as articulo_id,
                        '↳ ' + LTRIM(RTRIM(P.Descripcion)) as nombre,
                        P.Unidades as unidades,
                        0 as precio,
                        NULL as Hora,
                        '' as notas
                    FROM Comandas_Cabecera C WITH (NOLOCK)
                    INNER JOIN Comandas_Packs P WITH (NOLOCK) ON C.Numero_Referencia = P.Numero_Referencia AND C.Terminal = P.Terminal
                    WHERE P.Unidades <> 0

                    UNION ALL

                    SELECT
                        'DIRECTO-' + CAST(D.Numero AS VARCHAR) AS Numero_Referencia,
                        ISNULL(D.Mesa, 0) AS Mesa,
                        D.Nota_Cabecera1,
                        LTRIM(RTRIM(ISNULL(D.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        P.PLU as articulo_id,
                        '↳ ' + LTRIM(RTRIM(P.Descripcion)) as nombre,
                        P.Unidades as unidades,
                        0 as precio,
                        NULL as Hora,
                        '' as notas
                    FROM Documentos_Cabecera D WITH (NOLOCK)
                    INNER JOIN Documentos_Packs P WITH (NOLOCK) ON D.Serie = P.Serie AND D.Numero = P.Numero
                    WHERE P.Unidades <> 0
                    AND CAST(D.Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE))
                    AND D.Hora_Cierre >= DATEADD(minute, -15, GETDATE())
                    AND DATEDIFF(minute, D.Hora_Apertura, D.Hora_Cierre) <= 10
                `);

                const mapa = new Map();

                resVivas.recordset.forEach((row) => {
                    const idMesa = row.Numero_Referencia.toString();

                    if (!mapa.has(idMesa)) {
                        mapa.set(idMesa, {
                            numero_documento: idMesa,
                            mesa: row.Mesa.toString(),
                            notas_comanda: row.Nota_Cabecera1 || '',
                            nombre_cliente: '',
                            aperturaMs: null,
                            productosMap: new Map(),
                        });
                    }

                    const mesa = mapa.get(idMesa);
                    const nc = (row.Nombre_Cliente && String(row.Nombre_Cliente).trim()) || '';
                    if (nc && !mesa.nombre_cliente) mesa.nombre_cliente = nc;

                    if (row.Hora) {
                        const tm = new Date(row.Hora).getTime();
                        if (!Number.isNaN(tm)) {
                            if (mesa.aperturaMs == null || tm < mesa.aperturaMs) mesa.aperturaMs = tm;
                        }
                    }

                    const idPlato = row.articulo_id + '-' + row.nombre.trim();

                    if (!mesa.productosMap.has(idPlato)) {
                        mesa.productosMap.set(idPlato, {
                            articulo_id: row.articulo_id,
                            nombre: row.nombre.trim(),
                            unidades: 0,
                            precio: parseFloat(row.precio),
                            notas: row.notas.trim(),
                            hora: null,
                        });
                    }

                    const p = mesa.productosMap.get(idPlato);
                    p.unidades += parseFloat(row.unidades);
                    if (row.Hora) {
                        const tm = new Date(row.Hora).getTime();
                        if (!Number.isNaN(tm)) {
                            if (!p.hora || tm < new Date(p.hora).getTime()) {
                                p.hora = toIso(row.Hora);
                            }
                        }
                    }
                });

                const sala = Array.from(mapa.values())
                    .map((mesa) => {
                        const productosReales = Array.from(mesa.productosMap.values()).map((p) => ({
                            ...p,
                            numero_documento: mesa.numero_documento,
                        }));

                        return {
                            numero_documento: mesa.numero_documento,
                            id_ticket: mesa.numero_documento,
                            mesa: mesa.mesa,
                            notas_comanda: mesa.notas_comanda,
                            nombre_cliente: mesa.nombre_cliente || '',
                            timestamp_tpv:
                                mesa.aperturaMs != null
                                    ? toIso(mesa.aperturaMs)
                                    : toIso(new Date()),
                            productos: productosReales,
                        };
                    })
                    .filter((mesa) => mesa.productos.some((p) => p.unidades > 0));

                const estadoActual = JSON.stringify(sala);

                if (estadoActual !== ultimoEstadoSala) {
                    await axios.post(
                        URL_TELEMETRIA,
                        { sala, total_mesas_vivas: sala.length },
                        { timeout: AXIOS_TIMEOUT_MS }
                    );
                    process.stdout.write(
                        `\r[${new Date().toLocaleTimeString()}] 📡 Enviando a Cocina: ${sala.length} mesas vivas en curso.  `
                    );
                    ultimoEstadoSala = estadoActual;
                }
            } catch (e) {
                console.error('\nError Telemetría:', e.message);
            }
        }, 5000);
    } catch (err) {
        console.error('Error Crítico:', err.message);
    }
}

start();

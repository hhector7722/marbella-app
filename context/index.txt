const URL_TELEMETRIA = `${DOMINIO}/api/telemetria`;

/**
 * Ventana de negocio BDP (Data Sistema). Funciona aunque Fecha (día TPV) vaya desfasada.
 * El filtro solo por Hora_Cierre 3h devolvía 0 filas en producción (reloj/Hora_Cierre NULL).
 */
const VENTAS_WHERE = `
  WHERE (
    (Fecha_Sistema IS NOT NULL AND CAST(Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE)))
    OR (Fecha_Sistema IS NULL AND CAST(Fecha AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE)))
  )
`;

const CABECERA_SELECT = `
  Numero_Documento, Serie, Numero, Total_Documento,
  Fecha, Fecha_Sistema, Hora_Cierre,
  ISNULL(Mesa, 0) AS Mesa
`;

const CATCHUP_PAGE_SIZE = 100;
const POLL_INTERVAL_MS = 5000;
const AXIOS_TIMEOUT_MS = 45000;
const POLL_TOP = 50;

/** Solo historial al arrancar si BDP_RUN_CATCHUP=1. Por defecto OFF → solo poll (cobros nuevos). */
const RUN_CATCHUP = process.env.BDP_RUN_CATCHUP === '1';

let memoriaTickets = new Set();
let ultimoEstadoSala = "";
let ventasPollBusy = false;
let pollTick = 0;

function ticketKey(numeroDocumento) {
    return String(numeroDocumento ?? '').trim();
}

function toIso(val) {
    if (val == null) return null;
    const d = new Date(val);
    return Number.isNaN(d.getTime()) ? null : d.toISOString();
}
    
async function buildPayloadVenta(pool, t) {
    const lRes = await pool.request().input('s', t.Serie).input('n', t.Numero)
        .query("SELECT PLU, Descripcion, Unidades, Precio, Hora FROM Documentos_Lineas WITH (NOLOCK) WHERE Serie = @s AND Numero = @n");
    const mRes = await pool.request().input('s', t.Serie).input('n', t.Numero)
        .query("SELECT PLU, Descripcion, Unidades FROM Documentos_Menus WITH (NOLOCK) WHERE Serie = @s AND Numero = @n");
    const pRes = await pool.request().input('s', t.Serie).input('n', t.Numero)
        .query("SELECT PLU, Descripcion, Unidades FROM Documentos_Packs WITH (NOLOCK) WHERE Serie = @s AND Numero = @n");

    const horaLinea = lRes.recordset[0]?.Hora;

    return {
        numero_documento: ticketKey(t.Numero_Documento),
        mesa: t.Mesa,
        total_documento: parseFloat(t.Total_Documento) || 0,
        fecha_sistema: toIso(t.Fecha_Sistema),
        hora_cierre: toIso(t.Hora_Cierre),
        fecha_bdp: toIso(t.Fecha),
        fecha: horaLinea ? new Date(horaLinea).toISOString() : (toIso(t.Hora_Cierre) || new Date().toISOString()),
        lineas: [
            ...lRes.recordset.map(l => ({
                articulo_id: l.PLU,
                nombre: l.Descripcion?.trim() || "Sin Nombre",
                unidades: parseFloat(l.Unidades) || 0,
                precio: parseFloat(l.Precio) || 0
            })),
            ...mRes.recordset.map(m => ({
                articulo_id: m.PLU,
                nombre: '↳ ' + (m.Descripcion?.trim() || "Sin Nombre"),
                unidades: parseFloat(m.Unidades) || 0,
                precio: 0
            })),
            ...pRes.recordset.map(p => ({
                articulo_id: p.PLU,
                nombre: '↳ ' + (p.Descripcion?.trim() || "Sin Nombre"),
                unidades: parseFloat(p.Unidades) || 0,
                precio: 0
            }))
        ]
    };
}

async function enviarTicket(pool, t) {
    const key = ticketKey(t.Numero_Documento);
    const payload = await buildPayloadVenta(pool, t);
    await axios.post(URL_VENTAS, { ventas: [payload] }, { timeout: AXIOS_TIMEOUT_MS });
    memoriaTickets.add(key);
    console.log(`✅ Venta enviada -> Ticket: ${key} | Mesa: ${t.Mesa} | Fecha_Sistema: ${payload.fecha_sistema || '—'}`);
}

async function catchUpVentas(pool) {
    console.log('📦 Catch-up ventas (Fecha_Sistema, paginado)...');
    let offset = 0;
    let enviados = 0;
    let escaneados = 0;

    while (true) {
        const res = await pool.request()
            .input('off', sql.Int, offset)
            .input('lim', sql.Int, CATCHUP_PAGE_SIZE)
            .query(`
                SELECT ${CABECERA_SELECT}
                FROM Documentos_Cabecera WITH (NOLOCK)
                ${VENTAS_WHERE}
                ORDER BY Hora_Cierre DESC
                OFFSET @off ROWS FETCH NEXT @lim ROWS ONLY
            `);

        const batch = res.recordset || [];
        if (!batch.length) break;

        for (const t of batch) {
            escaneados++;
            if (memoriaTickets.has(ticketKey(t.Numero_Documento))) continue;
            await new Promise((r) => setTimeout(r, 50));
            try {
                await enviarTicket(pool, t);
                enviados++;
            } catch (e) {
                console.error(`❌ Catch-up ticket ${t.Numero_Documento}:`, e.message);
            }
        }

        offset += CATCHUP_PAGE_SIZE;
        if (batch.length < CATCHUP_PAGE_SIZE) break;
    }

    console.log(`📦 Catch-up listo: ${enviados} enviados (${escaneados} cabeceras en ventana).`);
}

async function logDiagnosticoSql(pool) {
    try {
        const diag = await pool.request().query(`
            SELECT
                GETDATE() AS sql_ahora,
                (SELECT COUNT(*) FROM Documentos_Cabecera WITH (NOLOCK) ${VENTAS_WHERE}) AS en_ventana,
                (SELECT COUNT(*) FROM Documentos_Cabecera WITH (NOLOCK)
                    WHERE Hora_Cierre IS NOT NULL AND Hora_Cierre >= DATEADD(hour, -3, GETDATE())) AS hora_cierre_3h,
                (SELECT MAX(Hora_Cierre) FROM Documentos_Cabecera WITH (NOLOCK)) AS max_hora_cierre,
                (SELECT MAX(Numero_Documento) FROM Documentos_Cabecera WITH (NOLOCK) ${VENTAS_WHERE}) AS max_doc_ventana
        `);
        const d = diag.recordset[0] || {};
        console.log('🔎 SQL GETDATE()      =', d.sql_ahora);
        console.log('🔎 Docs en ventana    =', d.en_ventana, '(Fecha_Sistema ayer/hoy)');
        console.log('🔎 Docs Hora_Cierre 3h =', d.hora_cierre_3h);
        console.log('🔎 MAX Hora_Cierre    =', d.max_hora_cierre);
        console.log('🔎 MAX Num.Documento  =', d.max_doc_ventana);
    } catch (e) {
        console.error('🔎 Diagnóstico SQL falló:', e.message);
    }
}

async function pollVentasRecientes(pool) {
    if (ventasPollBusy) return;
    ventasPollBusy = true;
    try {
        const res = await pool.request().query(`
            SELECT TOP (${POLL_TOP}) ${CABECERA_SELECT}
            FROM Documentos_Cabecera WITH (NOLOCK)
            ${VENTAS_WHERE}
            ORDER BY Hora_Cierre DESC, Numero_Documento DESC
        `);

        const candidatos = res.recordset || [];
        let nuevos = 0;

        for (const t of candidatos) {
            const key = ticketKey(t.Numero_Documento);
            if (memoriaTickets.has(key)) continue;
            try {
                await enviarTicket(pool, t);
                nuevos++;
            } catch (e) {
                console.error(`❌ ERROR ticket ${key}:`, e.message);
            }
        }

        pollTick++;
        const ts = new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
        if (nuevos > 0) {
            console.log(`🔄 [${ts}] Poll: ${nuevos} enviado(s) | candidatos=${candidatos.length} | memoria=${memoriaTickets.size}`);
        } else if (pollTick % 12 === 0) {
            console.log(`⏱ [${ts}] Poll: 0 nuevos | candidatos=${candidatos.length} | memoria=${memoriaTickets.size}`);
        }
    } finally {
        ventasPollBusy = false;
    }
}

async function start() {
    try {
        let pool = await sql.connect(config);
        console.log(`✅ Sistema Activado → ${DOMINIO}`);
        console.log(`📋 Poll cada ${POLL_INTERVAL_MS / 1000}s (Fecha_Sistema) | catch-up arranque: ${RUN_CATCHUP ? 'SÍ' : 'NO'}`);

        await logDiagnosticoSql(pool);

        void pollVentasRecientes(pool);

        setInterval(async () => {
            try {
                await pollVentasRecientes(pool);
            } catch (e) {
                console.error("❌ ERROR EN VENTAS:", e.message);
            }
        }, POLL_INTERVAL_MS);

        if (RUN_CATCHUP) {
            void catchUpVentas(pool).catch((e) => {
                console.error('❌ Catch-up en segundo plano:', e.message);
            });
        }

        // 2. TELEMETRÍA / MONITOR COCINA (CON CEREBRO MATEMÁTICO ANTIFRAUDE)
        //    Contrato JSON → Supabase `estado_sala.radiografia_completa` (Radar / KDS):
        //    - `timestamp_tpv`: ISO UTC de la **apertura más temprana** con `row.Hora` no nulo
        //    - Cada `producto` incluye `hora` (misma semántica) para que el dashboard pueda fallback.
        //    Ventas directas: filtro por Fecha_Sistema (alineado con ventas cobradas).
        setInterval(async () => {
            try {
                const resVivas = await pool.request().query(`
                    -- 1. MESAS VIVAS (Las normales de la sala)
                    SELECT 
                        CAST(C.Numero_Referencia AS VARCHAR) AS Numero_Referencia, 
                        ISNULL(C.Mesa, 0) AS Mesa, 
                        C.Nota_Cabecera1, 
                        LTRIM(RTRIM(ISNULL(C.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        L.PLU as articulo_id, 
                        L.Descripcion as nombre, 
                        L.Unidades as unidades, 
                        L.Precio as precio, 
                        L.Hora,
                        LTRIM(RTRIM(ISNULL(L.Descripcion_Auxiliar, '') + ISNULL((SELECT ' [' + CC.Descripcion + ']' FROM Comandas_Comentarios CC WITH (NOLOCK) WHERE CC.Numero_Referencia = L.Numero_Referencia AND CC.Terminal = L.Terminal AND CC.Linea_Comandas = L.Linea FOR XML PATH('')), ''))) as notas
                    FROM Comandas_Cabecera C WITH (NOLOCK)
                    INNER JOIN Comandas_Lineas L WITH (NOLOCK) ON C.Numero_Referencia = L.Numero_Referencia AND C.Terminal = L.Terminal
                    WHERE L.Unidades <> 0

                    UNION ALL

                    -- 2. VENTAS DIRECTAS NINJA (Cobradas al instante)
                    SELECT 
                        'DIRECTO-' + CAST(D.Numero AS VARCHAR) AS Numero_Referencia, 
                        ISNULL(D.Mesa, 0) AS Mesa, 
                        D.Nota_Cabecera1, 
                        LTRIM(RTRIM(ISNULL(D.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        L.PLU as articulo_id, 
                        L.Descripcion as nombre, 
                        L.Unidades as unidades, 
                        L.Precio as precio, 
                        L.Hora,
                        ISNULL(L.Descripcion_Auxiliar, '') as notas
                    FROM Documentos_Cabecera D WITH (NOLOCK)
                    INNER JOIN Documentos_Lineas L WITH (NOLOCK) ON D.Serie = L.Serie AND D.Numero = L.Numero
                    WHERE L.Unidades <> 0 
                    AND CAST(D.Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE))
                    AND D.Hora_Cierre >= DATEADD(minute, -15, GETDATE()) 
                    AND DATEDIFF(minute, D.Hora_Apertura, D.Hora_Cierre) <= 10

                    UNION ALL

                    -- 3. MENÚS (COMPONENTES)
                    SELECT 
                        CAST(C.Numero_Referencia AS VARCHAR) AS Numero_Referencia, 
                        ISNULL(C.Mesa, 0) AS Mesa, 
                        C.Nota_Cabecera1, 
                        LTRIM(RTRIM(ISNULL(C.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        M.PLU as articulo_id, 
                        '↳ ' + LTRIM(RTRIM(M.Descripcion)) as nombre, 
                        M.Unidades as unidades, 
                        0 as precio, 
                        NULL as Hora,
                        '' as notas
                    FROM Comandas_Cabecera C WITH (NOLOCK)
                    INNER JOIN Comandas_Menus M WITH (NOLOCK) ON C.Numero_Referencia = M.Numero_Referencia AND C.Terminal = M.Terminal
                    WHERE M.Unidades <> 0

                    UNION ALL

                    SELECT 
                        'DIRECTO-' + CAST(D.Numero AS VARCHAR) AS Numero_Referencia, 
                        ISNULL(D.Mesa, 0) AS Mesa, 
                        D.Nota_Cabecera1, 
                        LTRIM(RTRIM(ISNULL(D.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        M.PLU as articulo_id, 
                        '↳ ' + LTRIM(RTRIM(M.Descripcion)) as nombre, 
                        M.Unidades as unidades, 
                        0 as precio, 
                        NULL as Hora,
                        '' as notas
                    FROM Documentos_Cabecera D WITH (NOLOCK)
                    INNER JOIN Documentos_Menus M WITH (NOLOCK) ON D.Serie = M.Serie AND D.Numero = M.Numero
                    WHERE M.Unidades <> 0 
                    AND CAST(D.Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE))
                    AND D.Hora_Cierre >= DATEADD(minute, -15, GETDATE()) 
                    AND DATEDIFF(minute, D.Hora_Apertura, D.Hora_Cierre) <= 10

                    UNION ALL

                    -- 4. PACKS (COMPONENTES)
                    SELECT 
                        CAST(C.Numero_Referencia AS VARCHAR) AS Numero_Referencia, 
                        ISNULL(C.Mesa, 0) AS Mesa, 
                        C.Nota_Cabecera1, 
                        LTRIM(RTRIM(ISNULL(C.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        P.PLU as articulo_id, 
                        '↳ ' + LTRIM(RTRIM(P.Descripcion)) as nombre, 
                        P.Unidades as unidades, 
                        0 as precio, 
                        NULL as Hora,
                        '' as notas
                    FROM Comandas_Cabecera C WITH (NOLOCK)
                    INNER JOIN Comandas_Packs P WITH (NOLOCK) ON C.Numero_Referencia = P.Numero_Referencia AND C.Terminal = P.Terminal
                    WHERE P.Unidades <> 0

                    UNION ALL

                    SELECT 
                        'DIRECTO-' + CAST(D.Numero AS VARCHAR) AS Numero_Referencia, 
                        ISNULL(D.Mesa, 0) AS Mesa, 
                        D.Nota_Cabecera1, 
                        LTRIM(RTRIM(ISNULL(D.Nombre_Cliente, ''))) AS Nombre_Cliente,
                        P.PLU as articulo_id, 
                        '↳ ' + LTRIM(RTRIM(P.Descripcion)) as nombre, 
                        P.Unidades as unidades, 
                        0 as precio, 
                        NULL as Hora,
                        '' as notas
                    FROM Documentos_Cabecera D WITH (NOLOCK)
                    INNER JOIN Documentos_Packs P WITH (NOLOCK) ON D.Serie = P.Serie AND D.Numero = P.Numero
                    WHERE P.Unidades <> 0 
                    AND CAST(D.Fecha_Sistema AS DATE) >= DATEADD(day, -1, CAST(GETDATE() AS DATE))
                    AND D.Hora_Cierre >= DATEADD(minute, -15, GETDATE()) 
                    AND DATEDIFF(minute, D.Hora_Apertura, D.Hora_Cierre) <= 10
                `);

                let mapa = new Map();

                resVivas.recordset.forEach(row => {
                    const idMesa = row.Numero_Referencia.toString();

                    if (!mapa.has(idMesa)) {
                        mapa.set(idMesa, {
                            numero_documento: idMesa,
                            mesa: row.Mesa.toString(),
                            notas_comanda: row.Nota_Cabecera1 || '',
                            nombre_cliente: '',
                            aperturaMs: null,
                            productosMap: new Map()
                        });
                    }

                    const mesa = mapa.get(idMesa);
                    const nc = (row.Nombre_Cliente && String(row.Nombre_Cliente).trim()) || '';
                    if (nc && !mesa.nombre_cliente) mesa.nombre_cliente = nc;

                    if (row.Hora) {
                        const t = new Date(row.Hora).getTime();
                        if (!Number.isNaN(t)) {
                            if (mesa.aperturaMs == null || t < mesa.aperturaMs) mesa.aperturaMs = t;
                        }
                    }

                    const idPlato = row.articulo_id + '-' + row.nombre.trim();

                    if (!mesa.productosMap.has(idPlato)) {
                        mesa.productosMap.set(idPlato, {
                            articulo_id: row.articulo_id,
                            nombre: row.nombre.trim(),
                            unidades: 0,
                            precio: parseFloat(row.precio),
                            notas: row.notas.trim(),
                            hora: null
                        });
                    }

                    const p = mesa.productosMap.get(idPlato);
                    p.unidades += parseFloat(row.unidades);
                    if (row.Hora) {
                        const t = new Date(row.Hora).getTime();
                        if (!Number.isNaN(t)) {
                            if (!p.hora || t < new Date(p.hora).getTime()) p.hora = new Date(row.Hora).toISOString();
                        }
                    }
                });

                const sala = Array.from(mapa.values()).map(mesa => {
                    const productosReales = Array.from(mesa.productosMap.values()).map(p => ({
                        ...p,
                        numero_documento: mesa.numero_documento
                    }));

                    return {
                        numero_documento: mesa.numero_documento,
                        id_ticket: mesa.numero_documento,
                        mesa: mesa.mesa,
                        notas_comanda: mesa.notas_comanda,
                        nombre_cliente: mesa.nombre_cliente || '',
                        timestamp_tpv: mesa.aperturaMs != null
                            ? new Date(mesa.aperturaMs).toISOString()
                            : new Date().toISOString(),
                        productos: productosReales
                    };
                }).filter(mesa => mesa.productos.some(p => p.unidades > 0));

                const estadoActual = JSON.stringify(sala);

                if (estadoActual !== ultimoEstadoSala) {
                    await axios.post(URL_TELEMETRIA, { sala, total_mesas_vivas: sala.length });
                    process.stdout.write(`\r[${new Date().toLocaleTimeString()}] 📡 Enviando a Cocina: ${sala.length} mesas vivas en curso.  `);
                    ultimoEstadoSala = estadoActual;
                }

            } catch (e) { console.error('\nError Telemetría:', e.message);
            }
        }, 5000);

    } catch (err) { console.error('Error Crítico:', err.message); }
}
start();

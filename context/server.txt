const path = require('path');

require('dotenv').config({ path: path.join(__dirname, '.env') });
const express = require('express');
const cors = require('cors');
const { createClient } = require('@supabase/supabase-js');

const app = express();
app.use(cors());
app.use(express.json({ limit: '20mb' }));

const supabaseUrl = process.env.SUPABASE_URL;
const supabaseKey = process.env.SUPABASE_SERVICE_ROLE_KEY || process.env.SUPABASE_KEY;
const WEBHOOK_SECRET = process.env.WEBHOOK_SECRET || '';

/** fecha_sistema (default) | tpv (legacy: confiar en v.fecha del bridge) */
const VENTAS_FECHA_MODO = (process.env.VENTAS_FECHA_MODO || 'fecha_sistema').toLowerCase();
const TZ_MADRID = 'Europe/Madrid';

if (!supabaseUrl || !supabaseKey) {
    console.error("❌ ERROR CRÍTICO: No se han encontrado las variables de Supabase en el archivo .env");
    process.exit(1);
}

const supabase = createClient(supabaseUrl, supabaseKey);

function roundMoney(n) {
    return Math.round((Number(n) || 0) * 100) / 100;
}

function parseIso(value) {
    if (value == null || value === '') return null;
    const d = new Date(value);
    return Number.isNaN(d.getTime()) ? null : d;
}

function ymdMadrid(date) {
    return new Intl.DateTimeFormat('en-CA', {
        timeZone: TZ_MADRID,
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
    }).format(date);
}

function webhookAuthorized(req) {
    if (!WEBHOOK_SECRET) return true;
    return req.headers.authorization === `Bearer ${WEBHOOK_SECRET}`;
}

/**
 * Día contable desde Fecha_Sistema (Data Sistema BDP), no Fecha de negocio desfasada.
 * Body ventas: fecha_sistema, hora_cierre, fecha_bdp (auditoría), fecha (Hora línea, fallback).
 */
function resolveVentaTimestamps(v) {
    const receivedAt = new Date();

    if (VENTAS_FECHA_MODO === 'tpv') {
        const base = parseIso(v.fecha) || receivedAt;
        const cierre = parseIso(v.hora_cierre) || base;
        return {
            fecha: base.toISOString(),
            hora_cierre: cierre.toISOString(),
            fecha_real: receivedAt.toISOString(),
            diaNegocio: ymdMadrid(base),
        };
    }

    const instant =
        parseIso(v.fecha_sistema) ||
        parseIso(v.hora_cierre) ||
        parseIso(v.fecha) ||
        receivedAt;

    const horaCierre = parseIso(v.hora_cierre) || parseIso(v.fecha) || instant;

    return {
        fecha: instant.toISOString(),
        hora_cierre: horaCierre.toISOString(),
        fecha_real: receivedAt.toISOString(),
        diaNegocio: ymdMadrid(instant),
    };
}

/** Stock en segundo plano: el bridge no espera la RPC (evita colas de catch-up). */
function scheduleTicketStock(v) {
    const numeroDocStr = String(v.numero_documento);
    void (async () => {
        try {
            if (Number(v.total_documento) >= 0) {
                const { error: ledgerErr } = await supabase.rpc('process_ticket_stock_deduction', {
                    p_numero_documento: numeroDocStr,
                });
                if (ledgerErr) console.error(`[LEDGER] Deducción ${numeroDocStr}:`, ledgerErr.message);
            } else {
                const { error: refundErr } = await supabase.rpc('revert_ticket_stock_deduction', {
                    p_numero_documento: numeroDocStr,
                });
                if (refundErr) console.error(`[LEDGER] Reintegro ${numeroDocStr}:`, refundErr.message);
            }
        } catch (err) {
            console.error(`[LEDGER] Excepción ${numeroDocStr}:`, err.message);
        }
    })();
}

// =====================================
// FASE 1: VENTAS Y LÍNEAS DE TICKET
// =====================================
app.post('/api/ventas', async (req, res) => {
    try {
        const ventas = req.body.ventas || [];
        if (!ventas.length) return res.status(400).json({ error: "Vacío" });

        console.log(`[VENTAS] Recibidos ${ventas.length} tickets cerrados.`);

        for (const v of ventas) {
            const ts = resolveVentaTimestamps(v);
            const cobroEfectivo = roundMoney(v.cobro_efectivo);
            const cobroTarjeta = roundMoney(v.cobro_tarjeta);
            const cobroPendiente = roundMoney(v.cobro_pendiente);

            console.log(
                `[VENTAS] ${v.numero_documento} | modo=${VENTAS_FECHA_MODO} | ef=${cobroEfectivo} | tj=${cobroTarjeta} | pend=${cobroPendiente} | dia=${ts.diaNegocio}`
            );

            const { error: errCab } = await supabase.from('tickets_marbella').upsert([{
                numero_documento: v.numero_documento,
                mesa: v.mesa || 0,
                total_documento: v.total_documento,
                cobro_efectivo: cobroEfectivo,
                cobro_tarjeta: cobroTarjeta,
                cobro_pendiente: cobroPendiente,
                fecha: ts.fecha,
                hora_cierre: ts.hora_cierre,
                fecha_real: ts.fecha_real,
            }], { onConflict: 'numero_documento' });

            if (errCab) {
                console.error(`Error cabecera ${v.numero_documento}:`, errCab.message);
                continue;
            }

            if (v.lineas && v.lineas.length > 0) {
                const lineasTransformadas = v.lineas.map((l, index) => ({
                    numero_documento: v.numero_documento,
                    linea: index + 1,
                    articulo_id: l.articulo_id,
                    nombre: l.nombre,
                    unidades: l.unidades,
                    precio_unidad: l.precio,
                    importe_total: l.unidades * l.precio,
                    fecha_negocio: ts.fecha,
                    fecha_real: ts.fecha_real,
                }));

                const { error: errLin } = await supabase.from('ticket_lines_marbella').upsert(
                    lineasTransformadas,
                    { onConflict: 'numero_documento, linea' }
                );

                if (errLin) console.error(`Error líneas de ${v.numero_documento}:`, errLin.message);
            }

            scheduleTicketStock(v);
        }
        res.status(200).json({ success: true });
    } catch (err) {
        console.error("[VENTAS] Error general:", err.message);
        res.status(500).send(err.message);
    }
});

// =====================================
// MOVIMIENTOS DE CAJA BDP (staging bdp_cash_movements)
// =====================================
app.post('/api/caja', async (req, res) => {
    try {
        if (!webhookAuthorized(req)) {
            return res.status(401).json({ error: 'Unauthorized' });
        }

        const movimientos = req.body.movimientos || req.body.movements || [];
        if (!movimientos.length) return res.status(400).json({ error: 'Vacío' });

        const rows = movimientos
            .map((m) => {
                const amount = roundMoney(m.amount);
                const conceptCode = Number(m.concept_code);
                if (!m.movement_date || !m.fecha_negocio || !Number.isFinite(conceptCode) || amount === 0) {
                    return null;
                }
                return {
                    fecha_negocio: m.fecha_negocio,
                    movement_date: m.movement_date,
                    concept_code: conceptCode,
                    amount,
                    raw_json: m.raw_json ?? m,
                };
            })
            .filter(Boolean);

        if (!rows.length) return res.status(400).json({ error: 'Sin filas válidas' });

        const { error } = await supabase
            .from('bdp_cash_movements')
            .upsert(rows, {
                onConflict: 'movement_date,concept_code,amount',
                ignoreDuplicates: true,
            });

        if (error) {
            console.error('[CAJA] Error upsert:', error.message);
            return res.status(500).json({ error: error.message });
        }

        console.log(`[CAJA] ${rows.length} movimiento(s) procesados.`);
        res.status(200).json({ success: true, processed: rows.length });
    } catch (err) {
        console.error('[CAJA] Error general:', err.message);
        res.status(500).send(err.message);
    }
});

// =====================================
// FASE 2: TELEMETRÍA Y COMANDERO (KDS)
// Body: `{ total_mesas_vivas, sala }` — contrato en `context/index.txt`
// =====================================
app.post('/api/telemetria', async (req, res) => {
    console.log("\n====== NUEVO PAQUETE DE WINDOWS ======");
    console.log(JSON.stringify(req.body, null, 2));
    console.log("======================================\n");
    try {
        const { total_mesas_vivas, sala } = req.body;
        if (!sala) return res.status(200).send();

        const salaDisfrazada = sala.map(mesa => ({
            ...mesa,
            id_ticket: mesa.numero_documento,
            nombre_cliente:
                mesa.nombre_cliente != null && String(mesa.nombre_cliente).trim() !== ''
                    ? String(mesa.nombre_cliente).trim()
                    : '',
        }));

        const { error } = await supabase.from('estado_sala').upsert([{
            id: 1,
            ultima_actualizacion: new Date().toISOString(),
            mesas_activas: parseInt(total_mesas_vivas) || 0,
            radiografia_completa: salaDisfrazada,
        }], { onConflict: 'id' });

        if (error) throw error;

        console.log(`[RADAR] Sala y KDS actualizados: ${total_mesas_vivas} mesas activas.`);

        res.status(200).json({ success: true });
    } catch (err) {
        console.error("[TELEMETRIA] Error:", err.message);
        res.status(500).send(err.message);
    }
});

app.listen(3000, () => {
    console.log('----------------------------------------');
    console.log('🚀 RECEPTOR ERP BDP (PRODUCCIÓN ACTIVA)');
    console.log(`Modo fecha ventas: ${VENTAS_FECHA_MODO}`);
    console.log('Rutas: /api/ventas | /api/caja | /api/telemetria');
    console.log('Conectado a Supabase y escuchando en el puerto 3000');
    console.log('----------------------------------------');
});
